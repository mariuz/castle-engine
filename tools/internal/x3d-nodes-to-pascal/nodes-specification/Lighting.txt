# We add abstract X3DDirectionalLight, not present in X3D spec.
# This is an ancestor for X3D DirectionalLight and VRML 1.0 DirectionalLight.
X3DDirectionalLightNode : X3DPunctualLightNode {
  SFVec3f [in,out] direction        0 0 -1
    range: (-Inf,Inf)
    change: chLightLocationDirection

  # NOTE: global by default is FALSE

  # CASTLE GAME ENGINE EXTENSIONS:
  SFVec4f    [in,out]      projectionRectangle   0 0 0 0
    range: left, bottom, right, top; must be left < right and bottom < top
    # We want to set UpdateNeeded := true for all GeneratedShadowMap using
    # this light. For now, simply send chVisibleGeometry that will
    # force updating all GeneratedShadowMaps.
    change: chVisibleGeometry
    is-extension: true

  SFVec3f    [in,out]      projectionLocation    0 0 0
    # We want to set UpdateNeeded := true for all GeneratedShadowMap using
    # this light. For now, simply send chVisibleGeometry that will
    # force updating all GeneratedShadowMaps.
    change: chVisibleGeometry
    is-extension: true

  SFInt32    [in,out]      shadowCascades        1
    range: [1,4]
    change: chVisibleGeometry
    is-extension: true
    doc: """
    Number of cascades of the shadow map (when this light has @link(Shadows) by shadow maps).
    With 1 (the default), one shadow map covers the shadow receivers (or @link(ProjectionRectangle)).
    With 2 to 4, the view frustum of the main camera (TCastleAbstractRootTransform.MainCamera) is split
    into this many parts by the distance, each with its own shadow map
    (cascaded shadow maps): near the camera the shadows are sharper.
    The cascades follow the camera, they are rendered again every frame.
    Each cascade has the size of the @link(TGeneratedShadowMapNode.Size).
    See https://castle-engine.io/shadow_maps .
    """

  SFFloat    [in,out]      shadowCascadesDistance 0
    range: >= 0
    change: chVisibleGeometry
    is-extension: true
    doc: """
    How far from the camera the cascaded shadow maps reach (see @link(ShadowCascades)).
    0 (the default) means automatic: to the farthest point of the world's bounding box.
    Beyond it, nothing is shadowed by this light.
    """
}

# VRML 1.0 PointLight, VRML 1.0 SpotLight.

  # NOTE: global by default is TRUE

  SFVec3f [in,out] attenuation      1 0 0
    range: 0,Inf)
  SFVec3f [in,out] location         0 0 0
    range: 0 0 0 (-Inf,Inf)

  SFFloat [in,out] radius           100
    # As an extension to X3D, CGE allows radius -1 to indicate "no radius"
    range: [0,Inf) or -1
    change: chLightInstanceProperty
  SFBool [in,out] radiusFalloff     FALSE
    doc: Make the light fade smoothly to zero at its @link(Radius), instead of cutting it off there. The light's intensity (after @link(Attenuation)) is multiplied by @code(clamp(1 - (distance / radius)^4, 0, 1)^2), the window function recommended for the "range" of glTF @url(https://github.com/KhronosGroup/glTF/blob/main/extensions/2.0/Khronos/KHR_lights_punctual/README.md KHR_lights_punctual) lights. Ignored when the radius is <= 0 (no radius). This is a Castle Game Engine extension.
}

EnvironmentLight : X3DLightNode {
  SFNode  [in,out]  diffuseTexture   NULL
    range: X3DEnvironmentTextureNode
    doc: """
    Irradiance cube map: the diffuse light for each normal direction (in the light's local coordinates),
    multiplied by the material's diffuse color. Not used when @link(IrradianceVolume) is used.
    """
  SFNode  [in,out]  specularTexture  NULL
    range: X3DEnvironmentTextureNode
    doc: """
    Prefiltered radiance cube map: the light reflected in the direction of the reflected view
    (in the light's local coordinates). With physical materials, rougher surfaces use blurrier mipmap levels
    (the last level for roughness 1), so the mipmaps should contain the radiance convolved for increasing roughness.
    """

  # CASTLE GAME ENGINE EXTENSIONS:
  SFVec3f [in,out]  influenceCenter  0 0 0
    change: chVisibleNonGeometry
    is-extension: true
    doc: """
    Center of the box where this light (an environment probe) is used, in the light's local coordinates.
    See @link(InfluenceSize).
    """

  SFVec3f [in,out]  influenceSize    -1 -1 -1
    change: chVisibleNonGeometry
    is-extension: true
    doc: """
    Size of the box where this light (an environment probe) is used, in the light's local coordinates.
    A shape is lit by the environment light with the smallest box that contains the center
    of the shape's bounding box. Environment lights without a box (any size component < 0,
    the default) light only the shapes that are not inside any environment light's box:
    they are the fallback.
    So you can place many environment probes in a world, for example one for each room.
    """

  SFNode  [in,out]  irradianceVolume NULL
    range: X3DTexture3DNode
    is-extension: true
    doc: """
    Diffuse light that varies with the position: a grid of spherical harmonics (order 1, 4 coefficients for each color component)
    over the box of @link(InfluenceCenter) and @link(InfluenceSize), to use instead of @link(DiffuseTexture).

    The 3D texture holds RGB values. Its depth is 4 times the number of grid points along Z:
    first the slab of the constant coefficient (L0) of all grid points, then the slabs of the coefficients along X, Y and Z (L1),
    so a point (x, y, z) of the box (from 0 to 1) uses the texture coordinates (x, y, (z + i) / 4) for the coefficient i.
    The coefficients describe the irradiance (in light's local coordinates), so that the diffuse light for normal n
    is L0 + L1x * n.x + L1y * n.y + L1z * n.z (and this multiplies the material's diffuse color).
    Use linear filtering without mipmaps (@link(TTexturePropertiesNode.MinificationFilter) "AVG_PIXEL"),
    and a float format (e.g. a DDS volume texture with float components) for negative coefficients.
    """
}
