# We add abstract X3DDirectionalLight, not present in X3D spec.
# This is an ancestor for X3D DirectionalLight and VRML 1.0 DirectionalLight.
X3DDirectionalLightNode : X3DPunctualLightNode {
  SFVec3f [in,out] direction        0 0 -1
    range: (-Inf,Inf)
    change: chLightLocationDirection

  # NOTE: global by default is FALSE

  # CASTLE GAME ENGINE EXTENSIONS:
  SFVec4f    [in,out]      projectionRectangle   0 0 0 0
    range: left, bottom, right, top; must be left < right and bottom < top
    # We want to set UpdateNeeded := true for all GeneratedShadowMap using
    # this light. For now, simply send chVisibleGeometry that will
    # force updating all GeneratedShadowMaps.
    change: chVisibleGeometry
    is-extension: true

  SFVec3f    [in,out]      projectionLocation    0 0 0
    # We want to set UpdateNeeded := true for all GeneratedShadowMap using
    # this light. For now, simply send chVisibleGeometry that will
    # force updating all GeneratedShadowMaps.
    change: chVisibleGeometry
    is-extension: true

  SFInt32    [in,out]      shadowCascades        1
    range: [1,4]
    change: chVisibleGeometry
    is-extension: true
    doc: """
    Number of cascades of the shadow map (when this light has @link(Shadows) by shadow maps).
    With 1 (the default), one shadow map covers the shadow receivers (or @link(ProjectionRectangle)).
    With 2 to 4, the view frustum of the main camera (TCastleAbstractRootTransform.MainCamera) is split
    into this many parts by the distance, each with its own shadow map
    (cascaded shadow maps): near the camera the shadows are sharper.
    The cascades follow the camera, they are rendered again every frame.
    Each cascade has the size of the @link(TGeneratedShadowMapNode.Size).
    See https://castle-engine.io/shadow_maps .
    """

  SFFloat    [in,out]      shadowCascadesDistance 0
    range: >= 0
    change: chVisibleGeometry
    is-extension: true
    doc: """
    How far from the camera the cascaded shadow maps reach (see @link(ShadowCascades)).
    0 (the default) means automatic: to the farthest point of the world's bounding box.
    Beyond it, nothing is shadowed by this light.
    """
}

# We add abstract X3DPositionalLightNode, not present in X3D spec.
# This is an ancestor for at least X3D PointLight, X3D SpotLight,
# VRML 1.0 PointLight, VRML 1.0 SpotLight.
X3DPositionalLightNode : X3DPunctualLightNode {
  # NOTE: global by default is TRUE

  SFVec3f [in,out] attenuation      1 0 0
    range: 0,Inf)
  SFVec3f [in,out] location         0 0 0
    range: 0 0 0 (-Inf,Inf)
    change: chLightLocationDirection
  SFFloat [in,out] radius           100
    # As an extension to X3D, CGE allows radius -1 to indicate "no radius"
    range: [0,Inf) or -1
    change: chLightInstanceProperty
  SFBool [in,out] radiusFalloff     FALSE
    doc: Make the light fade smoothly to zero at its @link(Radius), instead of cutting it off there. The light's intensity (after @link(Attenuation)) is multiplied by @code(clamp(1 - (distance / radius)^4, 0, 1)^2), the window function recommended for the "range" of glTF @url(https://github.com/KhronosGroup/glTF/blob/main/extensions/2.0/Khronos/KHR_lights_punctual/README.md KHR_lights_punctual) lights. Ignored when the radius is <= 0 (no radius). This is a Castle Game Engine extension.
}
